# ============================================================
#  Makefile - Sistema de Gestión de Torneos
# ============================================================

# Compilador y estándar
CXX      := g++
CXXSTD   := -std=c++17

# Flags de compilación
CXXFLAGS := $(CXXSTD) -Wall -Wextra -g -MMD -MP
# Flags para el modo 'release' (make release)
RELEASE_FLAGS := $(CXXSTD) -O2 -DNDEBUG

# Carpetas del proyecto
SRC_DIR   := src
INC_DIR   := include
BUILD_DIR := build
BIN_DIR   := bin

# Incluye recursivamente todos los headers bajo include/
CPPFLAGS := -I$(INC_DIR)

# Nombre del ejecutable final
TARGET := $(BIN_DIR)/torneo

# Busca automáticamente todos los .cpp dentro de src/ (y subcarpetas)
SOURCES := $(shell find $(SRC_DIR) -name '*.cpp')

# Genera la ruta de los .o dentro de build/, preservando la subestructura de src/
OBJECTS := $(SOURCES:$(SRC_DIR)/%.cpp=$(BUILD_DIR)/%.o)

# Archivos de dependencias (para recompilar solo lo necesario si cambia un .hpp)
DEPS := $(OBJECTS:.o=.d)

# ------------------------------------------------------------
# Regla por defecto
# ------------------------------------------------------------
.PHONY: all
all: $(TARGET)

# Enlazado final
$(TARGET): $(OBJECTS) | $(BIN_DIR)
	$(CXX) $(CXXFLAGS) $(OBJECTS) -o $(TARGET)
	@echo "✅ Compilación completa: $(TARGET)"

# Compilación de cada .cpp a su .o correspondiente, creando subcarpetas si hace falta
$(BUILD_DIR)/%.o: $(SRC_DIR)/%.cpp | $(BUILD_DIR)
	@mkdir -p $(dir $@)
	$(CXX) $(CXXFLAGS) $(CPPFLAGS) -c $< -o $@

# Creación de carpetas de salida
$(BUILD_DIR):
	@mkdir -p $(BUILD_DIR)

$(BIN_DIR):
	@mkdir -p $(BIN_DIR)

# ------------------------------------------------------------
# Otros modos de compilación
# ------------------------------------------------------------

# Compilación optimizada para entrega/producción
.PHONY: release
release: CXXFLAGS := $(RELEASE_FLAGS)
release: clean all

# Ejecuta el programa tras compilarlo
.PHONY: run
run: all
	./$(TARGET)

# ------------------------------------------------------------
# Limpieza
# ------------------------------------------------------------
.PHONY: clean
clean:
	rm -rf $(BUILD_DIR) $(BIN_DIR)
	@echo "🧹 Archivos generados eliminados."

.PHONY: rebuild
rebuild: clean all

# ------------------------------------------------------------
# Incluye los archivos de dependencias generados (.d)
# ------------------------------------------------------------
-include $(DEPS)
